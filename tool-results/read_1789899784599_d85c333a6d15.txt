   313→/* ---------------- Open drawer ---------------- */
   314→
   315→function OpenDrawerDialog({
   316→  open,
   317→  onOpenChange,
   318→  symbol,
   319→  onOpened,
   320→}: {
   321→  open: boolean
   322→  onOpenChange: (o: boolean) => void
   323→  symbol: string
   324→  onOpened: () => void
   325→}) {
   326→  const { branches, activeBranchId } = useAuthStore()
   327→  const branchId = activeBranchId ?? branches[0]?.id
   328→  const [float, setFloat] = useState('5000')
   329→  const [note, setNote] = useState('')
   330→  const [busy, setBusy] = useState(false)
   331→
   332→  const submit = async () => {
   333→    const value = Number(float)
   334→    if (!Number.isFinite(value) || value < 0) {
   335→      toast.error('Enter the opening float — the cash you are starting the drawer with.')
   336→      return
   337→    }
   338→    setBusy(true)
   339→    try {
   340→      await api.post('/api/shifts', { branchId, openingFloat: value, note: note || null })
   341→      toast.success('Drawer is open', { description: `Started with ${formatMoney(value, symbol)}.` })
   342→      setNote('')
   343→      onOpenChange(false)
   344→      onOpened()
   345→    } catch (err) {
   346→      toast.error((err as Error).message)
   347→    } finally {
   348→      setBusy(false)
   349→    }
   350→  }
   351→
   352→  return (
   353→    <Dialog open={open} onOpenChange={onOpenChange}>
   354→      <DialogContent className="sm:max-w-sm" aria-describedby="open-drawer-desc">
   355→        <DialogHeader>
   356→          <DialogTitle className="flex items-center gap-2">
   357→            <Vault className="h-5 w-5 text-primary" /> Open the drawer
   358→          </DialogTitle>
   359→          <DialogDescription id="open-drawer-desc">
   360→            Count the cash you are starting with — it becomes the opening float for this shift.
   361→          </DialogDescription>
   362→        </DialogHeader>
   363→        <div className="space-y-3.5">
   364→          <div className="space-y-1.5">
   365→            <Label htmlFor="float-input">Opening float ({symbol})</Label>
   366→            <Input
   367→              id="float-input"
   368→              type="number"
   369→              min={0}
   370→              inputMode="decimal"
   371→              value={float}
   372→              onChange={(e) => setFloat(e.target.value)}
   373→              className="font-price h-11 text-lg"
   374→              autoFocus
   375→            />
   376→            <div className="flex gap-1.5 pt-1">
   377→              {FLOAT_CHIPS.map((v) => (
   378→                <button
   379→                  key={v}
   380→                  type="button"
   381→                  onClick={() => setFloat(String(v))}
   382→                  className={cn(
   383→                    'rounded-full border px-3 py-1 text-xs font-medium transition-colors',
   384→                    Number(float) === v ? 'border-primary bg-primary text-primary-foreground' : 'bg-card hover:bg-accent'
   385→                  )}
   386→                >
   387→                  {symbol} {v.toLocaleString()}
   388→                </button>
   389→              ))}
   390→            </div>
   391→          </div>
   392→          <div className="space-y-1.5">
   393→            <Label htmlFor="shift-note">Note (optional)</Label>
   394→            <Textarea
   395→              id="shift-note"
   396→              rows={2}
   397→              value={note}
   398→              onChange={(e) => setNote(e.target.value)}
   399→              placeholder="Morning shift, till 2…"
   400→            />
   401→          </div>
   402→          <Button className="w-full gap-2" onClick={submit} disabled={busy}>
   403→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Vault className="h-4 w-4" />}
   404→            Open drawer
   405→          </Button>
   406→        </div>
   407→      </DialogContent>
   408→    </Dialog>
   409→  )
   410→}

   412→/* ---------------- Count & close ---------------- */
   413→
   414→function CloseDrawerDialog({
   415→  shift,
   416→  onOpenChange,
   417→  symbol,
   418→  businessName,
   419→  branchName,
   420→  onClosed,
   421→}: {
   422→  shift: ShiftDto | null
   423→  onOpenChange: (o: boolean) => void
   424→  symbol: string
   425→  businessName: string
   426→  branchName: string
   427→  onClosed: () => void
   428→}) {
   429→  const [counted, setCounted] = useState('')
   430→  const [note, setNote] = useState('')
   431→  const [busy, setBusy] = useState(false)
   432→  const [closedReceipt, setClosedReceipt] = useState<ShiftDto | null>(null)
   433→
   434→  useEffect(() => {
   435→    if (shift) {
   436→      setCounted('')
   437→      setNote('')
   438→      setClosedReceipt(null)
   439→    }
   440→  }, [shift])
   441→
   442→  const expected = closedReceipt
   443→    ? closedReceipt.cashExpected ?? 0
   444→    : shift?.aggregates?.cashExpected ?? 0
   445→  const countedNum = Number(counted)
   446→  const variance =
   447→    counted.trim() !== '' && Number.isFinite(countedNum) ? Math.round((countedNum - expected) * 100) / 100 : null
   448→
   449→  const submit = async () => {
   450→    if (!shift) return
   451→    if (counted.trim() === '' || !Number.isFinite(countedNum) || countedNum < 0) {
   452→      toast.error('Enter the cash you counted in the drawer.')
   453→      return
   454→    }
   455→    setBusy(true)
   456→    try {
   457→      const closed = await api.post<ShiftDto>('/api/shifts/close', {
   458→        shiftId: shift.id,
   459→        countedCash: countedNum,
   460→        note: note || null,
   461→      })
   462→      setClosedReceipt(closed)
   463→      onClosed()
   464→    } catch (err) {
   465→      toast.error((err as Error).message)
   466→    } finally {
   467→      setBusy(false)
   468→    }
   469→  }
   470→
   471→  return (
   472→    <Dialog open={!!shift} onOpenChange={onOpenChange}>
   473→      <DialogContent className="sm:max-w-md" aria-describedby="close-drawer-desc">
   474→        {closedReceipt ? (
   475→          <>
   476→            <DialogHeader>
   477→              <DialogTitle className="flex items-center gap-2">
   478→                <ArrowDownToLine className="h-5 w-5 text-primary" /> Drawer closed
   479→              </DialogTitle>
   480→              <DialogDescription id="close-drawer-desc">
   481→                {varianceText(closedReceipt.variance ?? 0, symbol)} Print the close-out receipt for your records.
   482→              </DialogDescription>
   483→            </DialogHeader>
   484→            <div className="max-h-[46vh] overflow-y-auto scrollbar-thin rounded-xl border bg-white text-black">
   485→              <div className="receipt-print bg-white">
   486→                <ShiftReceipt shift={closedReceipt} businessName={businessName} branchName={branchName} symbol={symbol} />
   487→              </div>
   488→            </div>
   489→            <div className="flex justify-between gap-2">
   490→              <Button variant="outline" className="gap-2" onClick={() => window.print()}>
   491→                <Printer className="h-4 w-4" /> Print receipt
   492→              </Button>
   493→              <Button onClick={() => onOpenChange(false)}>Done</Button>
   494→            </div>
   495→          </>
   496→        ) : (
   497→          <>
   498→            <DialogHeader>
   499→              <DialogTitle className="flex items-center gap-2">
   500→                <ArrowDownToLine className="h-5 w-5 text-primary" /> Count the drawer
   501→              </DialogTitle>
   502→              <DialogDescription id="close-drawer-desc">
   503→                Count all cash in the drawer — notes and coins — and enter the total below.
   504→              </DialogDescription>
   505→            </DialogHeader>
   506→            <div className="rounded-xl border bg-muted/40 px-4 py-3">
   507→              <div className="flex items-center justify-between text-sm">
   508→                <span className="text-muted-foreground">Cash expected</span>
   509→                <span className="font-price text-lg font-bold">{formatMoney(expected, symbol)}</span>
   510→              </div>
   511→              <p className="mt-1 text-[11px] text-muted-foreground">
   512→                Float {formatMoney(shift?.openingFloat ?? 0, symbol)} + cash sales − payouts − returns
   513→              </p>
   514→            </div>
   515→            <div className="space-y-1.5">
   516→              <Label htmlFor="counted-input">Cash counted ({symbol})</Label>
   517→              <Input
   518→                id="counted-input"
   519→                type="number"
   520→                min={0}
   521→                inputMode="decimal"
   522→                value={counted}
   523→                onChange={(e) => setCounted(e.target.value)}
   524→                className="font-price h-11 text-lg"
   525→                autoFocus
   526→              />
   527→              {variance !== null && (
   528→                <p
   529→                  className={cn(
   530→                    'font-price rounded-lg px-3 py-2 text-sm font-medium',
   531→                    variance === 0
   532→                      ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300'
   533→                      : variance > 0
   534→                        ? 'bg-sky-50 text-sky-800 dark:bg-sky-950 dark:text-sky-300'
   535→                        : 'bg-red-50 text-destructive dark:bg-red-950'
   536→                  )}
   537→                >
   538→                  {variance === 0
   539→                    ? 'Perfect count — drawer is even.'
   540→                    : variance > 0
   541→                      ? `Over by ${formatMoney(variance, symbol)} — extra cash in the drawer.`
   542→                      : `Short by ${formatMoney(Math.abs(variance), symbol)} — cash is missing.`}
   543→                </p>
   544→              )}
   545→            </div>
   546→            <div className="space-y-1.5">
   547→              <Label htmlFor="close-note">Note (optional)</Label>
   548→              <Textarea
   549→                id="close-note"
   550→                rows={2}
   551→                value={note}
   552→                onChange={(e) => setNote(e.target.value)}
   553→                placeholder="Explain any variance…"
   554→              />
   555→            </div>
   556→            <div className="flex justify-end gap-2">
   557→              <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>
   558→                Cancel
   559→              </Button>
   560→              <Button className="gap-2" onClick={submit} disabled={busy}>
   561→                {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <ArrowDownToLine className="h-4 w-4" />}
   562→                Close drawer
   563→              </Button>
   564→            </div>
   565→          </>
   566→        )}
   567→      </DialogContent>
   568→    </Dialog>
   569→  )
   570→}
   571→
   572→function varianceText(variance: number, symbol: string) {
   573→  if (variance === 0) return 'Counted cash matches perfectly.'
   574→  if (variance > 0) return `Counted cash is over by ${formatMoney(variance, symbol)}.`
   575→  return `Counted cash is short by ${formatMoney(Math.abs(variance), symbol)}.`
   576→}
